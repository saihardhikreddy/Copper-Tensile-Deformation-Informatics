# LAMMPS Input Script for Tensile Deformation of Copper
# Written by: Sai Hardhik Reddy

# 1. INITIALIZATION
units           metal
dimension       3
boundary        p p p
atom_style      atomic

# 2. ATOM DEFINITION
lattice         fcc 3.615
region          simbox block 0 10 0 10 0 20
create_box      1 simbox
create_atoms    1 box

# 3. FORCE FIELD
pair_style      eam/alloy
# NOTE: You must have a Cu EAM file in your folder for this to work!
pair_coeff      * * Cu01.eam.alloy Cu

# 4. EQUILIBRATION (Relax the box at 900K)
velocity        all create 900.0 12345 dist gaussian
fix             1 all npt temp 900.0 900.0 0.1 iso 0.0 0.0 1.0
thermo          1000
run             5000
unfix           1

# 5. SETUP DEFORMATION VARIABLES
# Store initial length in Z direction as a static constant
variable        tmp equal lz
variable        L0 equal ${tmp}
# Calculate Engineering Strain: (L - L0) / L0
variable        strain equal "(lz - v_L0)/v_L0"
# Calculate Stress in Z direction (Convert from bar to GPa)
variable        stress_GPa equal "-pzz/10000"

# 6. DEFORMATION PROCESS
# Keep X and Y pressure at 0, hold Temp at 900K
fix             1 all npt temp 900.0 900.0 0.1 x 0.0 0.0 1.0 y 0.0 0.0 1.0
# Stretch the box in Z direction at a constant rate
fix             2 all deform 1 z erate 0.01 units box remap x

# 7. DATA OUTPUTS (For OVITO and Graphing)
# Compute the stress on every single atom
compute         atom_stress all stress/atom NULL

# Output 1: Dump file for OVITO (Now includes Z-direction atomic stress!)
dump            1 all custom 500 dump_900K.lammpstrj id type x y z c_atom_stress[3]

# Output 2: Text file for the Stress-Strain Curve (Excel/Python)
fix             def_print all print 100 "${strain} ${stress_GPa}" file stress_strain_900K.txt screen no

# 8. RUN THE STRETCHING
thermo          500
run             25000

print "SIMULATION COMPLETE! Check your folder for dump_900K.lammpstrj and stress_strain_900K.txt"
